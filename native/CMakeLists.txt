  target_link_libraries(luma_translate PRIVATE normaliz)
  target_compile_options(luma_translate PRIVATE /utf-8 /EHsc)
  add_custom_command(TARGET luma_translate POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/../windows/third_party/onnxruntime/onnxruntime.dll" "$<TARGET_FILE_DIR:luma_translate>/onnxruntime.dll")
else()
  target_link_libraries(luma_translate PRIVATE dl)
endif()
